export type SwingbyQuality='high'|'medium'|'low';
export interface SwingbySceneSnapshot {
 /** Actual runtime state, never a solution-preview epoch. */
 timeSeconds:number;positionKm:SwingbyVector;attitudeQuaternion:Quaternion;sunDirectionICRF:SwingbyVector;
 /** Caller-supplied visual surface rotation, not a body-fixed targeting model. */
 marsSurfaceRotationRad:number;
 objective:SwingbyObjectiveSnapshot;
 /** Actual domain body-frame thrust; renderer converts through the actual mount. */
 engine?:{partId:string;thrustDirectionBody:SwingbyVector};
}
export interface SwingbySceneCallbacks {

const MARS_RADIUS_KM=3389.5,SCALE=3/MARS_RADIUS_KM;
// Camera-only clearance, not a collision/success threshold. Near/contact states
// keep their exact position and remain renderable in the planetary overview.
const INSPECTION_CLEARANCE_KM=.02;
const FRAME=new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(1,0,0),-Math.PI/2);
const slots=['map','normalMap','roughnessMap','metalnessMap','emissiveMap','alphaMap','bumpMap','aoMap','lightMap','displacementMap','envMap'] as const;
export const swingbyScenePosition=(v:SwingbyVector)=>new THREE.Vector3(v[0]*SCALE,v[2]*SCALE,-v[1]*SCALE);
export const canInspectSwingbyState=(s:Pick<SwingbySceneSnapshot,'positionKm'>)=>new THREE.Vector3(...s.positionKm).length()>MARS_RADIUS_KM+INSPECTION_CLEARANCE_KM;
export function validateSwingbySceneSnapshot(s:SwingbySceneSnapshot){
 if(!s||!Number.isFinite(s.timeSeconds)||s.timeSeconds<0||!Number.isFinite(s.marsSurfaceRotationRad))throw new Error('Invalid actual swingby state.');
 for(const v of [s.positionKm,s.sunDirectionICRF])if(!Array.isArray(v)||v.length!==3||!v.every(Number.isFinite))throw new Error('Invalid swingby physical vector.');
 if(Math.abs(new THREE.Vector3(...s.sunDirectionICRF).length()-1)>1e-5)throw new Error('Sun direction must be a unit ICRF vector.');
 if(!Array.isArray(s.attitudeQuaternion)||s.attitudeQuaternion.length!==4||!s.attitudeQuaternion.every(Number.isFinite)||Math.abs(new THREE.Quaternion(...s.attitudeQuaternion).length()-1)>1e-5)throw new Error('Actual attitude must be a unit quaternion.');
 if(Math.abs(s.objective.elapsedSeconds-s.timeSeconds)>1e-6)throw new Error('Objective and actual spacecraft must share one display epoch.');
 if(s.engine&&(!s.engine.partId||!s.engine.thrustDirectionBody.every(Number.isFinite)||Math.abs(new THREE.Vector3(...s.engine.thrustDirectionBody).length()-1)>1e-5))throw new Error('Actual thrust must be a body-frame unit vector.');
}
