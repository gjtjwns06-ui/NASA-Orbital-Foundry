  async loadAssets() {
    const loader = new GLTFLoader(),
      tex = new THREE.TextureLoader();
    let done = 0;
    const total = 6;
    const loaded = (label: string) =>
      this.cb.assetProgress(label, ++done, total);
    await Promise.all([
      loader.loadAsync("/assets/launch-campus.glb").then((g) => {
        g.scene.traverse((o) => {
          if (o instanceof THREE.Mesh) {
            o.castShadow = true;
            o.receiveShadow = true;
          }
        });
        const palette: Record<string, string> = {
          Grass: "#527f4b",
          GrassLight: "#79a35e",
          TreeDark: "#235d4b",
          TreeMid: "#478252",
          TreeLight: "#7ca95d",
          Pink: "#c78eac",
          Teal: "#2b6373",
          Ivory: "#e2e5dc",
          ConcreteLight: "#b4beb1",
          Asphalt: "#394a50",
          Yellow: "#e4b24b",
          Amber: "#c97837",
        };
        g.scene.traverse((o) => {
          if (o instanceof THREE.Mesh) {
            const material = o.material as THREE.MeshStandardMaterial;
            if (palette[material.name])
              material.color.set(palette[material.name]);
          }
        });
        this.campus.add(g.scene);
        loaded("발사 기지");
      }),
      loader.loadAsync("/assets/hangar.glb").then((g) => {
        g.scene.traverse((o) => {
          if (o instanceof THREE.Mesh) {
            o.castShadow = true;
            o.receiveShadow = true;
          }
        });
        this.hangar.add(g.scene);
        loaded("조립실");
      }),
      tex.loadAsync("/assets/earth-day.jpg").then((t) => {
        t.colorSpace = THREE.SRGBColorSpace;
        t.anisotropy = 8;
        this.earthMap = t;
        this.earthMat.map = t;
        this.earthMat.needsUpdate = true;
        loaded("지구 표면");
      }),
      tex.loadAsync("/assets/mars.jpg").then((t) => {
        t.colorSpace = THREE.SRGBColorSpace;
        t.anisotropy = 8;
        this.marsMap = t;
        loaded("화성 표면");
      }),
      tex.loadAsync("/assets/earth-clouds.png").then((t) => {
        t.colorSpace = THREE.SRGBColorSpace;
        const m = this.cloudMesh.material as THREE.MeshStandardMaterial;
        m.map = t;
        m.opacity = 0.23;
        m.needsUpdate = true;
        this.cloudMesh.visible = this.mission === "earth";
        loaded("구름층");
      }),
      tex.loadAsync("/assets/earth-night.jpg").then((t) => {
        t.colorSpace = THREE.SRGBColorSpace;
        this.earthMat.emissiveMap = t;
        this.earthMat.needsUpdate = true;
        loaded("지구 야간");
      }),
    ]);
    this.ready = true;
  }

  setMode(mode: ViewMode, animate = true) {
    this.restoreFocusShadow();
    this.leaveControlRoom();
    this.followCraft = false;
    this.lastCraftPosition = null;
    this.mode = mode;
    this.selectedFunction.visible = mode === "build" && !this.stowOpen;
    this.setStowOpen(this.stowOpen);
    this.setBayOpen(this.bayOpen);
    if (this.inspectionGroup) this.inspectionGroup.visible = mode === "build";
    const ground = mode === "campus" || mode === "build";
    this.campus.visible = mode === "campus";
    this.hangar.visible = mode === "build";
    this.space.visible = !ground;
    this.craft.visible = mode !== "campus";
    this.bgSky.visible = mode === "campus";
    this.scene.background = new THREE.Color(
      mode === "build" ? 0x222f38 : 0x030912,
    );
    this.scene.fog = ground
      ? new THREE.Fog(
          mode === "build" ? 0x27333a : 0x94c2dc,
          mode === "build" ? 24 : 140,
          mode === "build" ? 60 : 340,
        )
      : null;
    this.hemi.intensity = ground ? 0.65 : 0.04;
    this.sun.intensity = ground ? 2.2 : 3;
    this.sun.position.set(
      ground ? 12 : 18,
      ground ? 18 : 3.24,
      ground ? 10 : -5.4,
    );
    this.sun.castShadow = ground;
    this.scene.environmentIntensity = ground ? 0.52 : 0.08;
    this.sun.shadow.camera.left = mode === "build" ? -9 : -26;
    this.sun.shadow.camera.right = mode === "build" ? 9 : 26;
    this.sun.shadow.camera.top = mode === "build" ? 9 : 26;
    this.sun.shadow.camera.bottom = mode === "build" ? -9 : -26;
    this.sun.shadow.camera.updateProjectionMatrix();
    this.controls.maxPolarAngle =
      mode === "build" ? Math.PI * 0.56 : ground ? Math.PI * 0.49 : Math.PI;
    this.controls.minDistance =
      mode === "build" ? 2.5 : mode === "campus" ? 15 : 3.4;
    this.controls.maxDistance =
      mode === "campus" ? 80 : mode === "build" ? 16 : 70;
    this.craft.scale.setScalar(ground ? 1 : 0.12);
    if (ground) this.craft.position.set(0, 1.8, 0);
    this.craft.quaternion.identity();

    this.rotationRing.visible = false;
    this.ao.enabled = ground && this.quality === "high";
    this.ao.kernelRadius = mode === "campus" ? 1.1 : 0.32;
    this.renderer.toneMappingExposure = ground ? 1.1 : 1.45;
    this.hemi.intensity = ground ? 0.65 : 0.13;
    const pos =
      mode === "campus"

      mode === "campus"
        ? new THREE.Vector3(-4, 4, 0)
        : mode === "build"
          ? new THREE.Vector3(0, 1.75, 0)

  flyCamera(pos: THREE.Vector3, target: THREE.Vector3, animate = true) {
    this.leaveControlRoom();
    if (animate)
      this.tween = {
        from: this.camera.position.clone(),
        to: pos,
        targetFrom: this.controls.target.clone(),
        target,
        t: 0,
      };
    else {
      this.camera.position.copy(pos);
      this.controls.target.copy(target);
      this.controls.update();
    }
  }
  setDesign(design: Design) {
    this.restoreFocusShadow();
    for (const marker of this.scienceMarkers.values()) {
      this.groundFeatures.remove(marker);
      disposeGeometry(marker);
    }
    this.scienceMarkers.clear();
    this.design = design;
    const busKey = design.hardwareVersion ? `${design.hardwareVersion}:${design.busId}` : "legacy";
    if (busKey !== this.busProfileKey) {
      this.craft.remove(this.bus);
      disposeGeometry(this.bus);
      this.bus=createBus(design);
      this.craft.add(this.bus);
      this.busProfileKey=busKey;
      this.setBayOpen(this.bayOpen);
    }
    const missionChanged = this.mission !== design.missionId;
    this.mission = design.missionId;
    this.scale = 3 / (this.mission === "earth" ? 6378.137 : 3389.5);
    this.earthMat.map = this.mission === "earth" ? this.earthMap : this.marsMap;
    this.earthMat.emissiveIntensity = this.mission === "earth" ? 1.2 : 0;
    this.earthMat.needsUpdate = true;
    this.cloudMesh.visible =
      this.mission === "earth" &&
      !!(this.cloudMesh.material as THREE.MeshStandardMaterial).map;
    const atm = this.planet.getObjectByName("atmosphere") as THREE.Mesh;
    (atm.material as THREE.ShaderMaterial).uniforms.tint.value.set(
      this.mission === "earth" ? 0x329ec7 : 0xc77749,
    );
    for (const [id, o] of this.parts) {
      if (!design.parts.some((p) => p.id === id)) {
        this.craft.remove(o);
        disposeGeometry(o);
        this.parts.delete(id);
      }
    }
    for (const p of design.parts) {
      let o = this.parts.get(p.id);
      const modelKey=`${design.hardwareVersion || "legacy"}:${p.kind}:${p.modelId || "legacy"}`;
      if (o && o.userData.modelKey !== modelKey) {
        this.craft.remove(o);
        disposeGeometry(o);
        this.parts.delete(p.id);
        o = undefined;
      }
      if (!o) {
        o = createPart(p,design);
        o.userData.partKind = p.kind;
        o.userData.modelKey = modelKey;
        o.traverse((c) => (c.userData.partId = p.id));
        this.parts.set(p.id, o);
        this.craft.add(o);
      }
      this.placeObject(o, p);
    }
    if (this.stowOpen) this.setStowOpen(true);
    this.setSelected(this.selection);
    if (this.inspectionKind) this.setInspection(this.inspectionKind);
    if (missionChanged) {
      clearGroup(this.referencePath);
      if (this.mode === "plan") this.setMode("plan");
    }
  }
  async enterControlRoom() {
    if (this.mode !== "flight") return;
    this.roomRequested = true;
    if (!this.controlRoom)
      this.controlRoom = new ControlRoom(this.canvas, this.scene.environment);
    await this.controlRoom.load();
    if (!this.roomRequested || this.mode !== "flight") return;
    this.tween = null;
    this.controls.enabled = false;
    this.controlRoom.controls.enabled = true;
    this.controlRoomActive = true;
  }
  leaveControlRoom() {
    this.roomRequested = false;
    this.controlRoomActive = false;
    this.controls.enabled = true;
    if (this.controlRoom) this.controlRoom.controls.enabled = false;
  }

    );

        this.mission === "earth"
          ? earthStateAt(this.design.plan, time)

      this.camera.position.add(movement);

    }

    this.craft.position.copy(p);
    if (attitude) {
      const q = new THREE.Quaternion(...attitude).premultiply(
        new THREE.Quaternion().setFromAxisAngle(
          new THREE.Vector3(1, 0, 0),
          -Math.PI / 2,

  private onDown(e: PointerEvent) {
    if (e.button !== 0) return;
    if (this.drag && this.ghost) return;
    this.down = { x: e.clientX, y: e.clientY };
    this.coordinates(e.clientX, e.clientY);
    if (this.mode === "build") {
      if (this.stowOpen) return;
      if (this.inspectionGroup) {
        const sunHits = this.ray
          .intersectObject(this.inspectionGroup, true)
          .filter((h) => h.object.userData.inspectionHandle === "sun");
        if (sunHits.length) {
          this.drag = {
            type: "nav",
            handle: "inspection-sun",
            x: e.clientX,
            y: e.clientY,
            moved: false,
          };
          this.controls.enabled = false;
          return;
        }
      }
      const rotate = this.rotationRing.visible
        ? this.ray.intersectObject(this.rotationRing)
        : [];
      if (rotate.length && this.selection) {
        this.drag = {
          type: "nav",
          handle: "rotate",
          x: e.clientX,
          y: e.clientY,
          moved: false,
        };
        this.controls.enabled = false;
        return;
      }
      const hits = this.ray.intersectObjects([...this.parts.values()], true);
      if (hits.length) {
        const id = hits[0].object.userData.partId as string;
        this.cb.select(id);
        const part = this.design.parts.find((p) => p.id === id)!;
        this.drag = {
          type: "part",
          part: { ...part },
          x: e.clientX,
          y: e.clientY,
          moved: false,
        };
        this.controls.enabled = false;
      } else this.cb.select(null);
    } else if (this.mode === "plan") {
      if (this.targetPicking) return;
      const canvasRect = this.canvas.getBoundingClientRect();
      const candidates = this.handleGroup.children
        .filter((o) => o.userData.handle)
        .map((o) => {
          const p = this.project(o.getWorldPosition(new THREE.Vector3()));
          return {
            o,
            d: Math.hypot(
              p.x + canvasRect.left - e.clientX,
              p.y + canvasRect.top - e.clientY,
            ),
            visible: p.visible,
          };
        })
        .filter((c) => c.visible && c.d < 23)
        .sort((a, b) => a.d - b.d);
      if (candidates.length) {
        const handle = candidates[0].o.userData.handle as string;
        if (handle.startsWith("node:")) {
          this.selectedManeuverId = handle.slice(5);
          this.cb.node(this.selectedManeuverId);
          this.down = null;
          return;
        }
        this.drag = {
          type: "nav",
          handle,
          x: e.clientX,
          y: e.clientY,
          moved: false,
        };
        this.controls.enabled = false;
        return;
      }
    }
  }
  private onMove(e: PointerEvent) {
    if (!this.drag) return;
    const d = this.drag,
      dx = e.clientX - d.x,
      dy = e.clientY - d.y;
    if (Math.abs(dx) + Math.abs(dy) > 2) d.moved = true;
    if (d.type === "part") {
      if (!d.moved) return;
      if (!this.ghost) {
        this.ghostPart = d.part!;
        this.ghost = createPart(d.part!,this.design);
        this.ghost.traverse((o) => {
          if (o instanceof THREE.Mesh) {
            const m = (o.material as THREE.Material).clone();
            m.transparent = true;
            m.opacity = 0.55;
            m.userData.ghostMaterial = true;
            o.material = m;
          }
        });
        this.craft.add(this.ghost);
        const original = this.parts.get(d.part!.id);
        if (original) original.visible = false;
      }
      this.updatePlacement(e.clientX, e.clientY);
    } else {
      const factor = e.shiftKey ? 0.2 : 1;
      this.cb.navigation(d.handle!, dx * factor, dy * factor, false);
      d.x = e.clientX;
      d.y = e.clientY;
    }
  }
  private updatePlacement(x: number, y: number) {
    if (!this.ghost || !this.ghostPart) return;
    this.coordinates(x, y);
    this.mount.updateMatrixWorld(true);
    const hits = this.ray.intersectObject(this.mount);
    if (!hits.length) {
      this.ghost.visible = false;
      return;
    }
    this.ghost.visible = true;
    const hit = hits[0],
      local = this.craft.worldToLocal(hit.point.clone()),
      n = hit.face!.normal;
    let face: Face =
      Math.abs(n.x) > 0.5
        ? n.x > 0
          ? "px"
          : "nx"
        : Math.abs(n.y) > 0.5
          ? n.y > 0
            ? "py"
            : "ny"
          : n.z > 0
            ? "pz"
            : "nz";
    const dummy = { ...this.ghostPart, face, u: 0, v: 0 };
    const tr = partTransformFor(this.design,{ ...dummy, rotation: 0 }),
      delta = local.sub(new THREE.Vector3(...tr.position));
    dummy.u = delta.dot(new THREE.Vector3(...tr.tangentU));
    dummy.v = delta.dot(new THREE.Vector3(...tr.tangentV));
    this.ghostPart = dummy;
    this.placeObject(this.ghost, dummy);
    const valid = this.cb.placement(dummy, false);
    this.ghost.traverse((o) => {
      if (o instanceof THREE.Mesh) {
        const m = o.material as THREE.MeshStandardMaterial;
        if (m.emissive) m.emissive.set(valid ? 0x005449 : 0x861f13);
      }
    });
  }
  private onUp(e: PointerEvent) {
    if (
      this.targetPicking &&
      this.mode === "plan" &&
      this.mission === "earth" &&
      this.down &&
      Math.hypot(e.clientX - this.down.x, e.clientY - this.down.y) < 5
    ) {
      this.coordinates(e.clientX, e.clientY);
      const hit = this.ray.intersectObject(this.earthMesh)[0];
      if (hit) {
        const p = this.earthMesh.worldToLocal(hit.point.clone()).normalize();
        this.cb.target(
          (Math.asin(p.y) * 180) / Math.PI,
          (Math.atan2(-p.z, p.x) * 180) / Math.PI,
        );
        this.down = null;
        return;
      }
    }
    const d = this.drag;
    if (d) {
      if (d.type === "part" && d.moved && this.ghost?.visible && this.ghostPart)
        this.cb.placement(this.ghostPart, true);
      if (d.type === "nav") this.cb.navigation(d.handle!, 0, 0, true);
      this.cancelDrag();
    } else if (
      this.mode === "plan" &&
      this.mission === "earth" &&
      this.down &&
      Math.hypot(e.clientX - this.down.x, e.clientY - this.down.y) < 5
    ) {
      const r = this.canvas.getBoundingClientRect();
      let best = 25,
        time = -1;
      for (const p of this.trajectoryPoints) {
        const v = v3(p.positionKm)
          .multiplyScalar(this.scale)
          .project(this.camera);
        if (v.z > 1) continue;
        const dist = Math.hypot(
          (v.x * 0.5 + 0.5) * r.width + r.left - e.clientX,
          (-v.y * 0.5 + 0.5) * r.height + r.top - e.clientY,
        );
        if (dist < best) {
          best = dist;
          time = p.t;
        }
      }
      if (time >= 0) this.cb.burnAt(time);
    }
    this.down = null;
  }

  render(dt: number) {
    if (this.restoreEnvironment) {
      const pmrem = new THREE.PMREMGenerator(this.renderer),
        room = new RoomEnvironment();
      const next = pmrem.fromScene(room, 0.04);
      room.dispose();
      pmrem.dispose();
      this.environmentTarget = next;
      this.environmentTargetDisposed=false;
      this.scene.environment = next.texture;
      this.restoreEnvironment = false;
    }
    this.canvas.style.cursor =
      this.targetPicking && this.mode === "plan"
        ? "crosshair"
        : this.drag
          ? "grabbing"
          : "";
    this.elapsed += dt;
    const now = performance.now();
    this.frameTimes.push(now - this.lastFrame);
    this.lastFrame = now;
    if (this.frameTimes.length > 7200) this.frameTimes.shift();
    if (this.inspectionGroup)
      this.inspectionGroup.visible = this.mode === "build";
    if (this.launchSequence) {
      const s = this.launchSequence;
      s.time += dt;
      const rise = Math.pow(Math.max(0, s.time - 1.6), 2) * 2.8;
      s.rocket.position.y = s.base.y + rise;
      s.effects.position.set(s.base.x, s.base.y + rise + 0.2, s.base.z);
      s.effects.children
        .slice(0, 2)
        .forEach((o, i) =>
          o.scale.setScalar(
            (0.9 + Math.sin(s.time * 50 + i) * 0.1) * Math.min(1, s.time),
          ),
        );
      s.effects.children.slice(2).forEach((o, i) => {
        const phase = (s.time * 0.6 + i * 0.06) % 1,
          a = i * 2.399;
        o.position.set(
          Math.cos(a) * phase * 7,
          -rise + phase * 0.8,
          Math.sin(a) * phase * 7,
        );
        o.scale.setScalar(0.6 + phase * 2);
        (o as THREE.Mesh).material instanceof THREE.MeshStandardMaterial &&
          ((
            o as THREE.Mesh<THREE.BufferGeometry, THREE.MeshStandardMaterial>
          ).material.opacity = (1 - phase) * 0.65);
      });
      this.controls.target.y = 6 + rise * 0.65;
      if (s.time > 5.6) this.skipLaunch();
    }
    if (this.tween) {
      this.tween.t = Math.min(1, this.tween.t + dt * 1.6);
      const a = this.tween.t,
        e = a * a * (3 - 2 * a);
      this.camera.position.lerpVectors(this.tween.from, this.tween.to, e);
      this.controls.target.lerpVectors(
        this.tween.targetFrom,
        this.tween.target,
        e,
      );
      if (a === 1) this.tween = null;
    }
    if (this.mode === "campus" && !this.launchSequence)
      this.controls.autoRotate = true;
    else this.controls.autoRotate = false;
    this.controls.autoRotateSpeed = 0.16;
    this.controls.update(dt);
    if (this.followCraft && this.camera.position.length() < 3.09)
      this.camera.position.normalize().multiplyScalar(3.09);
    this.handleGroup.children.forEach((o) => {
      if (o.userData.handle) {
        o.quaternion.copy(this.camera.quaternion);
      }
    });
    if (this.outline.visible) this.outline.update();
    this.renderer.info.reset();
    if (this.controlRoomActive && this.controlRoom?.ready)
      this.controlRoom.render(this.renderer, this.scene, this.camera, dt);
    else this.composer.render(dt);
  }
  setQuality(q: "high" | "medium" | "low") {

  }

function disposeGeometry(o: THREE.Object3D) {
  const materials = new Set<THREE.Material>();
  const textures = new Set<THREE.Texture>();
  o.traverse((c) => {
    if (c instanceof THREE.Mesh) {
      c.geometry.dispose();
      for (const m of Array.isArray(c.material) ? c.material : [c.material])
        if (!isSharedPartMaterial(m)) materials.add(m);
    }
  });
  for (const m of materials) {
    for(const key of ["map","bumpMap","normalMap","roughnessMap","metalnessMap","emissiveMap","alphaMap"] as const) {
      const texture=(m as THREE.MeshStandardMaterial)[key];
      if(texture&&!isSharedPartTexture(texture))textures.add(texture);
    }
    m.dispose();
  }
  for(const texture of textures)texture.dispose();
}
function clearGroup(g: THREE.Group) {
  for (const child of [...g.children]) {
    g.remove(child);
    child.traverse((o) => {
      if (o instanceof THREE.Mesh || o instanceof THREE.Line) {
        o.geometry.dispose();
        for (const m of Array.isArray(o.material) ? o.material : [o.material]) {
          const map = (m as THREE.MeshBasicMaterial).map;
          if (map) map.dispose();
          m.dispose();
        }
      }
    });
  }
