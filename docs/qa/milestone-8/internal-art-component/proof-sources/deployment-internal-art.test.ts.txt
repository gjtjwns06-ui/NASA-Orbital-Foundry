import {describe,it,expect} from 'vitest';
import * as THREE from 'three';
import {createHash} from 'node:crypto';
import {HARDWARE_MODELS} from '../src/hardware/catalog';
import {compileDeploymentCraft,createDeploymentCraft} from '../src/deployment/craft';
import {initialJointStates,resolveDeploymentGeometry} from '../src/deployment/kinematics';
import {DeploymentView} from '../src/render/deployment-view';
import {createDeploymentInternal} from '../src/render/deployment-internal-art';
import {createDeploymentLink,deploymentPalette} from '../src/render/deployment-link-meshes';
import {createDeploymentLink as baselineLink,deploymentPalette as baselinePalette} from '../src/render/deployment-link-meshes-baseline';
import type {LocalBox} from '../src/deployment/types';
const tally=(o:THREE.Object3D)=>{let triangles=0,batches=0;o.traverse(c=>{if(c instanceof THREE.Mesh){batches++;triangles+=(c.geometry.index?.count??c.geometry.attributes.position.count)/3;}});return {triangles,batches};};
const dispose=(o:THREE.Object3D,p:ReturnType<typeof deploymentPalette>)=>{o.traverse(c=>{if(c instanceof THREE.Mesh)c.geometry.dispose();});Object.values(p).forEach(m=>m.dispose());};
const fixture=createDeploymentCraft(),compiled=compileDeploymentCraft(fixture),topology=compiled.topology.definition;
const signature=(o:THREE.Object3D)=>{const h=createHash('sha256');o.traverse(c=>{if(c instanceof THREE.Mesh){for(const name of Object.keys(c.geometry.attributes)){const a=c.geometry.getAttribute(name) as THREE.BufferAttribute;h.update(Buffer.from(a.array.buffer,a.array.byteOffset,a.array.byteLength));}h.update(JSON.stringify(c.geometry.index?.array??null));const m=c.material as THREE.MeshStandardMaterial;h.update(JSON.stringify([m.color.toArray(),m.metalness,m.roughness,m.polygonOffset,m.polygonOffsetFactor,m.polygonOffsetUnits]));}});return h.digest('hex');};
describe('bounded original commissioning internal art',()=>{
 it('contains every vertex inside its exact authoritative box including quaternion-oriented boxes',()=>{const models=Object.values(HARDWARE_MODELS).filter(m=>m.internal&&/^(bpx|iris|ax100|xact|recorder)/.test(m.modelId));const boxes=models.map(m=>({variant:m.modelId,box:{id:'proxy',centerM:m.geometry.deployedBoxes[0].center,halfM:m.geometry.deployedBoxes[0].half,renderRole:'electronics'} as LocalBox}));boxes.push({variant:'mechanism-controller-v1',box:{id:'body',centerM:[0,0,.025],halfM:[.06,.04,.025],renderRole:'electronics'}});for(const {variant,box} of boxes)for(const rotated of[false,true]){const b={...box,quaternion:rotated?new THREE.Quaternion().setFromAxisAngle(new THREE.Vector3(1,2,3).normalize(),.71).toArray() as [number,number,number,number]:undefined},p=deploymentPalette(),g=createDeploymentInternal(b,variant,p)!;expect(g,variant).not.toBeNull();g.updateMatrixWorld(true);const iq=b.quaternion?new THREE.Quaternion(...b.quaternion).invert():new THREE.Quaternion();g.traverse(o=>{if(o instanceof THREE.Mesh){const a=o.geometry.attributes.position;for(let i=0;i<a.count;i++){const v=new THREE.Vector3().fromBufferAttribute(a,i).applyMatrix4(o.matrixWorld).sub(new THREE.Vector3(...b.centerM)).applyQuaternion(iq);v.toArray().forEach((n,j)=>expect(Math.abs(n),`${variant}/${j}`).toBeLessThanOrEqual(b.halfM[j]+2e-8));}}});dispose(g,p);}});
 it('keeps external bus, solar/RF/support mesh output byte-identical',()=>{for(const link of topology.links.filter(l=>l.render.family!=='electronics')){const pa=deploymentPalette(),pb=baselinePalette(),a=createDeploymentLink(link,topology,pa),b=baselineLink(link,topology,pb);expect(signature(a),link.id).toBe(signature(b));dispose(a,pa);dispose(b,pb);}});
 it('stays within four batches per default internal link and the6000 extra triangle budget',()=>{let extra=0;const stats=[];for(const link of topology.links.filter(l=>l.render.family==='electronics')){const pa=deploymentPalette(),pb=baselinePalette(),a=createDeploymentLink(link,topology,pa),b=baselineLink(link,topology,pb),after=tally(a),before=tally(b);stats.push({id:link.id,before,after});expect(after.batches,link.id).toBeLessThanOrEqual(4);extra+=after.triangles-before.triangles;dispose(a,pa);dispose(b,pb);}expect(stats).toHaveLength(5);expect(extra).toBeLessThanOrEqual(6000);console.log(JSON.stringify({stats,extra}));});
 it('preserves immutable topology/mass/FK and keeps part-level ray picking',()=>{const before=JSON.stringify(topology),geometry=resolveDeploymentGeometry(compiled.topology,initialJointStates(compiled.topology)),physical=JSON.stringify(geometry);for(const link of topology.links.filter(l=>l.render.family==='electronics')){const p=deploymentPalette(),g=createDeploymentLink(link,topology,p);g.updateMatrixWorld(true);const hits=new THREE.Raycaster(new THREE.Vector3(0,0,2),new THREE.Vector3(0,0,-1)).intersectObject(g,true);expect(hits.length,link.id).toBeGreaterThan(0);expect(g.userData.partId).toBe(link.partId);dispose(g,p);}expect(JSON.stringify(topology)).toBe(before);expect(JSON.stringify(resolveDeploymentGeometry(compiled.topology,initialJointStates(compiled.topology)))).toBe(physical);});
 it('makes actual/ghost resources instance-owned and final disposal idempotent without new textures',()=>{const a=new DeploymentView(compiled.topology),b=new DeploymentView(compiled.topology),view=resolveDeploymentGeometry(compiled.topology,initialJointStates(compiled.topology));a.setGeometryView(view);b.setGeometryView(view);a.setPreviewGeometry(view);const resources=new Set<THREE.BufferGeometry|THREE.Material>(),other=new Set<THREE.BufferGeometry|THREE.Material>();for(const [root,set]of[[a.actualRoot,resources],[a.previewRoot,resources],[b.actualRoot,other]]as const)root.traverse(o=>{if(o instanceof THREE.Mesh){set.add(o.geometry);set.add(o.material as THREE.Material);expect((o.material as THREE.MeshStandardMaterial).map??null).toBeNull();}});const counts=new Map<unknown,number>();for(const r of resources){expect(other.has(r)).toBe(false);r.addEventListener('dispose',()=>counts.set(r,(counts.get(r)??0)+1));}let touchedOther=0;other.forEach(r=>r.addEventListener('dispose',()=>touchedOther++));a.dispose();a.dispose();expect([...resources].every(r=>counts.get(r)===1)).toBe(true);expect(touchedOther).toBe(0);b.dispose();});
});
