async function witness(p:Page,inspect=true){
 if(inspect)await control(p,'sweep').click();await expect(p.locator('.deployment-clearance-cue:not([hidden])')).toHaveCount(2);
 const s=await state(p),w=s.preview.report.firstWitness??s.preview.report.firstUncertainWitness;
 const facts=await p.locator('.deployment-clearance-cue button').evaluateAll(es=>es.map(el=>{const b=el as HTMLButtonElement,r=b.getBoundingClientRect(),dot=b.parentElement!.querySelector('.deployment-cue-dot')!.getBoundingClientRect(),hit=document.elementFromPoint(r.x+r.width/2,r.y+r.height/2);return {data:{...b.dataset},text:b.textContent,rect:{left:r.left,top:r.top,right:r.right,bottom:r.bottom},anchor:{x:dot.x+dot.width/2,y:dot.y+dot.height/2},hit:b===hit||b.contains(hit)};}));
 const projections=await p.evaluate(()=>{const d=(window as any).__game.deployment(),s=d.getState();return d.scene.getClearanceCues(s.draft.inputHash,s.preview.requestId);});
 for(const [i,f]of facts.entries()){const expected=i===0?w.a:w.b;expect(f.data.partId).toBe(expected.partId);expect(f.data.linkId).toBe(expected.linkId);expect(f.data.boxId).toBe(expected.boxId);expect(f.data.sourceId).toBe(s.draft.inputHash);expect(f.data.requestId).toBe(s.preview.requestId);expect(f.hit).toBe(true);expect(f.anchor.x).toBeCloseTo(projections.points[i].x,1);expect(f.anchor.y).toBeCloseTo(projections.points[i].y,1);expect(f.rect.left).toBeGreaterThanOrEqual(projections.bounds.left);expect(f.rect.right).toBeLessThanOrEqual(projections.bounds.right);expect(f.rect.bottom).toBeLessThanOrEqual(projections.bounds.bottom);}
 writeFileSync(join(out,`witness-${p.viewportSize()!.width}-${s.preview.requestId}.json`),JSON.stringify({facts,projections,witness:w},null,2));return s;
}
